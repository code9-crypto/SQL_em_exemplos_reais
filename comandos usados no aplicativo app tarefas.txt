CRIAÇÃO DE PROCEDURES PARA MELHORAR APLICAÇÃO NAS BUSCAS E INSERÇÕES

//Esta PROCEDURE é para buscar as tarefas pelo nome do solicitante
DELIMITER $$

CREATE PROCEDURE recuperaTarefaPeloNome(
    IN p_nome VARCHAR(100)
)
BEGIN

    SELECT
        tf_data,
        tf_destino,
        tf_solicitante,
        tf_tarefa
    FROM tarefas
    WHERE tf_solicitante LIKE CONCAT('%', p_nome, '%');

END $$

DELIMITER ;

-------------------------------------------------------------------
//Esta procedure traz as regiões de 1 a 9

delimiter $$

create procedure recuperaRegioes()

BEGIN
	
    SELECT
    	DISTINCT(tf_destino)
        	FROM tarefas
        	WHERE tf_destino LIKE "%Região%";

END $$

DELIMITER ;

----------------------------------------------------------------------
//Este procedure busca pelas tarefas por meio da região
DELIMITER $$

create PROCEDURE recuperaTarefaPorRegiao(
	in p_regiao varchar(50)
)

BEGIN
	
    select tf_data, tf_destino, tf_solicitante, tf_tarefa from tarefas
    where tf_destino like concat("%", p_regiao, "%");

END $$

DELIMITER ;

--------------------------------------------------------------------------
//Esta procedure recupera as tarefas por meio da data
DELIMITER $$

CREATE PROCEDURE recuperaTarefaPorData(
    IN p_data VARCHAR(100)
)
BEGIN

    SELECT
        tf_data,
        tf_destino,
        tf_solicitante,
        tf_tarefa
    FROM tarefas
    WHERE tf_data = p_data

END $$

DELIMITER ;

--------------------------------------------------------------------------
//Esta procedure recupera uma única tarefa por meio do número
DELIMITER $$

CREATE PROCEDURE retornaTarefaPorNumero(
    IN p_numero VARCHAR(100)
)
BEGIN

    SELECT
        tf_data,
        tf_destino,
        tf_solicitante,
        tf_tarefa
    FROM tarefas
    WHERE tf_tarefa = p_numero

END $$

DELIMITER ;


------------------------------------------------------------------------------
//Esta procedure irá salvar os dados no banco
delimiter $$

create procedure gravaTarefa(
	in p_tarefa varchar(10),
    in p_solicitante varchar(150),
    in p_destino varchar (50)
)

BEGIN
	insert into tarefas(
    	tf_tarefa,
        tf_solicitante,
        tf_destino
    )
    
    VALUES (
    	p_tarefa,
        p_solicitante,
        p_destino
    );
    
END $$

DELIMITER ;


----------------------------------------------------
//Esta procedure vai fazer a busca pelo SSHD do solicitante por meio do seu nome
DELIMITER $$

create procedure buscaSSHD(

    IN b_nome varchar(200)

)
BEGIN
	select sshd_solicitante
    FROM sshd_solicitantes
    	WHERE b_nome = nome_solicitante
        LIMIT 1;
END

DEMILITER ;


-----------------------------------------------------
//Esta procedure irá salvar o nome e o sshd do solicitante na tabela sshd_solicitantes


DELIMITER $$

create procedure salvaSSHD(

    IN s_nome varchar(200),
    IN s_sshd varchar(10)
        
)

BEGIN
	INSERT INTO sshd_solicitantes(
        nome_solicitante,
        sshd_solicitante
    )
    
    VALUES(
        s_nome,
        s_sshd
   	);
END

DELIMITER ;
